val reminderFormDateFormatter: DateTimeFormatter =
    DateTimeFormatter.ofPattern("dd/MM/yyyy")

val reminderFormTimeFormatter: DateTimeFormatter =
    DateTimeFormatter.ofPattern("HH:mm")

fun isReminderDateTimeOnOrAfter(
    laterDate: String,
    laterTime: String,
    earlierDate: String,
    earlierTime: String
): Boolean {

    return try {

        val later =
            LocalDateTime.of(
                LocalDate.parse(
                    laterDate,
                    reminderFormDateFormatter
                ),
                LocalTime.parse(
                    laterTime,
                    reminderFormTimeFormatter
                )
            )

        val earlier =
            LocalDateTime.of(
                LocalDate.parse(
                    earlierDate,
                    reminderFormDateFormatter
                ),
                LocalTime.parse(
                    earlierTime,
                    reminderFormTimeFormatter
                )
            )

        !later.isBefore(earlier)

    } catch (exception: Exception) {

        false
    }
}

fun isReminderDateOnOrAfter(
    laterDate: String,
    earlierDate: String
): Boolean {

    return try {

        val later =
            LocalDate.parse(
                laterDate,
                reminderFormDateFormatter
            )

        val earlier =
            LocalDate.parse(
                earlierDate,
                reminderFormDateFormatter
            )

        !later.isBefore(earlier)

    } catch (exception: Exception) {

        false
    }
}

                                "Deadline"

                                when {

                                        .isNotBlank() -> {

                                        selectedRepeatUntilDate
                                    }

                                    selectedType ==
                                            "Deadline" -> {

                                        selectedDeadlineDate
                                    }

                                    else -> {

                                        "None"
                                    }
                                }

                            } else {

                                "None"
                            }

                        val anchorDate =
                            if (recurring) {
                                selectedDate
                            } else {
                                "None"
                            }

                        val reminder =
                            Reminder(
                                title =
                                    title.trim(),
                                type =
                                    selectedType,
                                date =
                                    selectedDate,
                                time =
                                    selectedTime,
                                priority =
                                    selectedPriority,
                                category =
                                    selectedCategory,
                                reminderTiming =
                                    serializeReminderTimings(
                                        selectedReminderTimings
                                    ),
                                notes =
                                    notes.trim(),
                                repeatRule =
                                    selectedRepeatRule,
                                repeatAnchorDate =
                                    anchorDate,
                                deadlineDate =
                                    finalDeadlineDate,
                                deadlineTime =
                                    finalDeadlineTime,
                                repeatUntilDate =
                                    repeatUntil,
                                escalationEnabled =
                                    selectedType ==
                                        "Deadline" &&
                                        automaticEscalationEnabled
                            )

                        coroutineScope.launch {

                            val newId =
                                database
                                    .reminderDao()
                                    .insertReminder(
                                        reminder
                                    )

                            val savedReminder =
                                reminder.copy(
                                    id =
                                        newId.toInt()
                                )

                            val scheduled =
                                ReminderScheduler
                                    .scheduleReminder(
                                        context,
                                        savedReminder
                                    )

                            if (scheduled) {

                                Toast.makeText(
                                    context,
                                    "Reminder saved and scheduled",
                                    Toast.LENGTH_SHORT
                                ).show()

                            } else {

                                Toast.makeText(
                                    context,
                                    "Reminder saved, but no future notification could be scheduled.",
                                    Toast.LENGTH_LONG
                                ).show()
                            }

                            onBack()
                        }
                    }
                }
            },
            modifier =
                Modifier
                    .fillMaxWidth()
                    .height(54.dp),
            shape =
                RoundedCornerShape(16.dp)
        ) {

            Text(
                text = "Save reminder"
            )
        }
    }
}

@Composable
fun AutomaticEscalationToggle(
    enabled: Boolean,
    priority: String,
    onEnabledChange: (Boolean) -> Unit
) {

    val description =
        when (priority) {

            "Critical" ->
                "Adds intensive deadline alerts: 1 week, 3 days, 1 day, 2 hours, 1 hour, 30 minutes, 10 minutes and at the deadline."

            "Important" ->
                "Adds balanced deadline alerts: 3 days, 1 day, 2 hours, 30 minutes and at the deadline."

            else ->
                "Adds gentle deadline alerts: 1 day, 1 hour and at the deadline."
        }

    Row(
        modifier =
            Modifier.fillMaxWidth(),
        verticalAlignment =
            Alignment.CenterVertically
    ) {

        Column(
            modifier =
                Modifier.weight(1f)
        ) {

            Text(
                text =
                    "Automatic escalation",
                style =
                    MaterialTheme
                        .typography
                        .titleMedium
            )

            Spacer(
                modifier =
                    Modifier.height(4.dp)
            )

            Text(
                text =
                    description,
                style =
                    MaterialTheme
                        .typography
                        .bodyMedium,
                color =
                    MaterialTheme
                        .colorScheme
                        .onSurfaceVariant
            )
        }

        Spacer(
            modifier =
                Modifier.size(14.dp)
        )

        Switch(
            checked =
                enabled,
            onCheckedChange =
                onEnabledChange
        )
    }
}

@Composable
fun SimpleDropdown(
    selectedValue: String,
    options: List<String>,
    onValueSelected: (String) -> Unit
) {

    var expanded by remember {
        mutableStateOf(false)
    }

    Box(
        modifier =
            Modifier.fillMaxWidth()
    ) {

        OutlinedButton(
            onClick = {
                expanded = true
            },
            modifier =
                Modifier
                    .fillMaxWidth()
                    .height(52.dp),
            shape =
                RoundedCornerShape(14.dp)
        ) {

            Text(
                text = selectedValue
            )
        }

        DropdownMenu(
            expanded = expanded,
            onDismissRequest = {
                expanded = false
            }
        ) {

            options.forEach { option ->

                DropdownMenuItem(
                    text = {
                        Text(
                            text = option
                        )
                    },
                    onClick = {

                        onValueSelected(
                            option
                        )

                        expanded = false
                    }
                )
            }
        }
    }
}

                            selectedRepeatRule
                        ) &&
                            selectedRepeatUntilDate
                                .isNotBlank() &&
                            !isReminderDateOnOrAfter(
                                selectedRepeatUntilDate,
                                selectedDate
                            ) -> {

                        Toast.makeText(
                            context,
                            "Repeat Until cannot be before the next reminder date.",
                            Toast.LENGTH_LONG
                        ).show()
                    }

                    selectedType ==
                            "Deadline" &&
                            selectedRepeatUntilDate

                        ).show()

                    else -> {

                        val recurring =
                            ReminderScheduler
                                .isRecurring(
                                    selectedRepeatRule
                                )

                        val updatedDeadlineDate =
                            if (
                                selectedType ==
                                "Deadline"
                            ) {
                                selectedDeadlineDate
                            } else {
                                "None"
                            }

                        val updatedDeadlineTime =
                            if (
                                selectedType ==
                                "Deadline"
                            ) {
                                selectedDeadlineTime
                            } else {
                                "None"
                            }

                        val updatedRepeatUntil =
                            if (recurring) {

                                when {

                                    selectedRepeatUntilDate
                                        .isNotBlank() -> {

                                        selectedRepeatUntilDate
                                    }

                                    selectedType ==
                                            "Deadline" -> {

                                        selectedDeadlineDate
                                    }

                                    else -> {

                                        "None"
                                    }
                                }

                            } else {

                                "None"
                            }

                        val updatedAnchorDate =
                            when {

                                !recurring -> {

                                    "None"
                                }

                                !ReminderScheduler
                                    .isRecurring(
                                        reminder.repeatRule
                                    ) -> {

                                    selectedDate
                                }

                                selectedDate !=
                                        reminder.date -> {

                                    selectedDate
                                }

                                selectedRepeatRule !=
                                        reminder.repeatRule -> {

                                    selectedDate
                                }

                                reminder.repeatAnchorDate !=
                                        "None" -> {

                                    reminder.repeatAnchorDate
                                }

                                else -> {

                                    selectedDate
                                }
                            }

                        val updatedReminder =
                            reminder.copy(
                                title =
                                    title.trim(),
                                type =
                                    selectedType,
                                date =
                                    selectedDate,
                                time =
                                    selectedTime,
                                priority =
                                    selectedPriority,
                                category =
                                    selectedCategory,
                                reminderTiming =
                                    serializeReminderTimings(
                                        selectedReminderTimings
                                    ),
                                notes =
                                    notes.trim(),
                                repeatRule =
                                    selectedRepeatRule,
                                repeatAnchorDate =
                                    updatedAnchorDate,
                                deadlineDate =
                                    updatedDeadlineDate,
                                deadlineTime =
                                    updatedDeadlineTime,
                                repeatUntilDate =
                                    updatedRepeatUntil,
                                escalationEnabled =
                                    selectedType ==
                                        "Deadline" &&
                                        automaticEscalationEnabled
                            )

                        coroutineScope.launch {

                            ReminderScheduler
                                .cancelReminder(
                                    context,
                                    reminder.id
                                )

                            database
                                .reminderDao()
                                .updateReminder(
                                    updatedReminder
                                )

                            val scheduled =
                                ReminderScheduler
                                    .scheduleReminder(
                                        context,
                                        updatedReminder
                                    )

                            if (scheduled) {

                                Toast.makeText(
                                    context,
                                    "Reminder updated and scheduled",
                                    Toast.LENGTH_SHORT
                                ).show()

                Modifier

object ReminderScheduler {

    private val dateFormatter =
        DateTimeFormatter.ofPattern("dd/MM/yyyy")

    private val timeFormatter =
        DateTimeFormatter.ofPattern("HH:mm")

    private val timingCodes = mapOf(
        "At time of event" to 1,
        "10 minutes before" to 2,
        "30 minutes before" to 3,
        "1 hour before" to 4,
        "2 hours before" to 5,
        "1 day before" to 6,
        "3 days before" to 7,
        "1 week before" to 8
    )

    /*
     * Deadline escalation alarms use their own request-code
     * range so they never overwrite normal occurrence alarms.
     */
    private val escalationTimingCodes = mapOf(
        "At time of event" to 21,
        "10 minutes before" to 22,
        "30 minutes before" to 23,
        "1 hour before" to 24,
        "2 hours before" to 25,
        "1 day before" to 26,
        "3 days before" to 27,
        "1 week before" to 28
    )

    private const val RECURRENCE_CODE = 90

    fun scheduleReminder(
        context: Context,
        reminder: Reminder
    ): Boolean {

        if (reminder.time.isBlank()) {
            return false
        }

        val eventDateTime =
            parseEventDateTime(reminder)
                ?: return false

        /*
         * A recurring reminder must never schedule an
         * occurrence after its Repeat Until date.
         */
        if (
            isRecurring(reminder.repeatRule) &&
            !isOccurrenceWithinRepeatLimit(reminder)
        ) {
            return false
        }

        val selectedTimings =
            parseReminderTimings(
                reminder.reminderTiming
            )

        val deadlineDateTime =
            if (
                reminder.escalationEnabled &&
                reminder.type.equals(
                    "Deadline",
                    ignoreCase = true
                )
            ) {
                parseDeadlineDateTime(
                    reminder
                )
            } else {
                null
            }

        val escalationTimings =
            if (deadlineDateTime != null) {
                getEscalationTimings(
                    reminder.priority
                )
            } else {
                emptyList()
            }

        val currentOccurrenceIsDeadline =
            deadlineDateTime != null &&
                eventDateTime ==
                deadlineDateTime

        var scheduledAtLeastOneNotification =
            false

        /*
         * Keep the user's manually selected notification
         * timings. On the final deadline occurrence, skip
         * only timings that automatic escalation will
         * already provide so the user does not receive
         * duplicate notifications at the same moment.
         */
        selectedTimings.forEach { timing ->

            val coveredByEscalation =
                currentOccurrenceIsDeadline &&
                    timing in escalationTimings

            if (!coveredByEscalation) {

                val notificationDateTime =
                    calculateNotificationTime(
                        eventDateTime,
                        timing
                    )

                val triggerTime =
                    notificationDateTime
                        .atZone(
                            ZoneId.systemDefault()
                        )
                        .toInstant()
                        .toEpochMilli()

                if (
                    triggerTime >
                    System.currentTimeMillis()
                ) {

                    scheduleSingleNotificationAlarm(
                        context = context,
                        reminder = reminder,
                        timing = timing,
                        triggerTime = triggerTime
                    )

                    scheduledAtLeastOneNotification =
                        true
                }
            }
        }

        /*
         * Automatic escalation is calculated from the
         * final deadline, not from the next recurring
         * occurrence. The user's priority controls how
         * intensive the extra alerts become.
         */
        if (deadlineDateTime != null) {

            escalationTimings.forEach { timing ->

                val notificationDateTime =
                    calculateNotificationTime(
                        deadlineDateTime,
                        timing
                    )

                val triggerTime =
                    notificationDateTime
                        .atZone(
                            ZoneId.systemDefault()
                        )
                        .toInstant()
                        .toEpochMilli()

                if (
                    triggerTime >
                    System.currentTimeMillis()
                ) {

                    scheduleSingleDeadlineEscalationAlarm(
                        context = context,
                        reminder = reminder,
                        timing = timing,
                        triggerTime = triggerTime
                    )

                    scheduledAtLeastOneNotification =
                        true
                }
            }
        }

        if (
            isRecurring(reminder.repeatRule) &&
            hasNextOccurrence(reminder)
        ) {

            val eventTriggerTime =
                eventDateTime
                    .atZone(
                        ZoneId.systemDefault()
                    )
                    .toInstant()
                    .toEpochMilli()

            if (
                eventTriggerTime >
                System.currentTimeMillis()
            ) {

                scheduleRecurrenceAlarm(
                    context = context,
                    reminder = reminder,
                    triggerTime = eventTriggerTime
                )
            }
        }

        return scheduledAtLeastOneNotification
    }

    private fun scheduleSingleNotificationAlarm(
        context: Context,
        reminder: Reminder,
        timing: String,
        triggerTime: Long
    ) {

        val notificationId =
            createAlarmId(
                reminder.id,
                timing
            )

        val message =
            if (reminder.notes.isBlank()) {

                "$timing • ${reminder.type} scheduled " +
                        "for ${reminder.date} at ${reminder.time}"

            } else {

                "$timing • ${reminder.notes}"
            }

        val intent =
            Intent(
                context,
                ReminderReceiver::class.java
            ).apply {

                putExtra(
                    "REMINDER_ID",
                    reminder.id
                )

                putExtra(
                    "NOTIFICATION_ID",
                    notificationId
                )

                putExtra(
                    "REMINDER_TITLE",
                    reminder.title
                )

                putExtra(
                    "REMINDER_MESSAGE",
                    message
                )

                putExtra(
                    "REMINDER_REPEAT_RULE",
                    reminder.repeatRule
                )

                putExtra(
                    "OCCURRENCE_DATE",
                    reminder.date
                )
            }

        val pendingIntent =
            PendingIntent.getBroadcast(
                context,
                notificationId,
                intent,
                PendingIntent.FLAG_UPDATE_CURRENT or
                        PendingIntent.FLAG_IMMUTABLE
            )

        scheduleAlarm(
            context = context,
            triggerTime = triggerTime,
            pendingIntent = pendingIntent
        )
    }

    private fun scheduleSingleDeadlineEscalationAlarm(
        context: Context,
        reminder: Reminder,
        timing: String,
        triggerTime: Long
    ) {

        val notificationId =
            createEscalationAlarmId(
                reminder.id,
                timing
            )

        val timingText =
            when (timing) {

                "At time of event" ->
                    "Deadline reached"

                else ->
                    timing
            }

        val message =
            "$timingText • Final deadline " +
                "${reminder.deadlineDate} at " +
                reminder.deadlineTime

        val intent =
            Intent(
                context,
                ReminderReceiver::class.java
            ).apply {

                putExtra(
                    "REMINDER_ID",
                    reminder.id
                )

                putExtra(
                    "NOTIFICATION_ID",
                    notificationId
                )

                putExtra(
                    "REMINDER_TITLE",
                    reminder.title
                )

                putExtra(
                    "REMINDER_MESSAGE",
                    message
                )

                /*
                 * Escalation notifications refer to the whole
                 * deadline, not one recurring occurrence.
                 * Therefore "Done" completes the deadline
                 * reminder instead of advancing one week.
                 */
                putExtra(
                    "REMINDER_REPEAT_RULE",
                    "None"
                )

                putExtra(
                    "OCCURRENCE_DATE",
                    reminder.deadlineDate
                )
            }

        val pendingIntent =
            PendingIntent.getBroadcast(
                context,
                notificationId,
                intent,
                PendingIntent.FLAG_UPDATE_CURRENT or
                    PendingIntent.FLAG_IMMUTABLE
            )

        scheduleAlarm(
            context = context,
            triggerTime = triggerTime,
            pendingIntent = pendingIntent
        )
    }

    private fun getEscalationTimings(
        priority: String
    ): List<String> {

        return when (priority) {

            "Critical" ->
                listOf(
                    "1 week before",
                    "3 days before",
                    "1 day before",
                    "2 hours before",
                    "1 hour before",
                    "30 minutes before",
                    "10 minutes before",
                    "At time of event"
                )

            "Important" ->
                listOf(
                    "3 days before",
                    "1 day before",
                    "2 hours before",
                    "30 minutes before",
                    "At time of event"
                )

            else ->
                listOf(
                    "1 day before",
                    "1 hour before",
                    "At time of event"
                )
        }
    }

    private fun scheduleRecurrenceAlarm(
        context: Context,
        reminder: Reminder,
        triggerTime: Long
    ) {

        val recurrenceAlarmId =
            createRecurrenceAlarmId(
                reminder.id
            )

        val intent =
            Intent(
                context,
                RecurrenceReceiver::class.java
            ).apply {

                putExtra(
                    "REMINDER_ID",
                    reminder.id
                )
            }

        val pendingIntent =
            PendingIntent.getBroadcast(
                context,
                recurrenceAlarmId,
                intent,
                PendingIntent.FLAG_UPDATE_CURRENT or
                        PendingIntent.FLAG_IMMUTABLE
            )

        scheduleAlarm(
            context = context,
            triggerTime = triggerTime,
            pendingIntent = pendingIntent
        )
    }

    private fun scheduleAlarm(
        context: Context,
        triggerTime: Long,
        pendingIntent: PendingIntent
    ) {

        val alarmManager =
            context.getSystemService(
                Context.ALARM_SERVICE
            ) as AlarmManager

        if (
            Build.VERSION.SDK_INT >=
            Build.VERSION_CODES.S
        ) {

            if (
                alarmManager.canScheduleExactAlarms()
            ) {

                alarmManager.setExactAndAllowWhileIdle(
                    AlarmManager.RTC_WAKEUP,
                    triggerTime,
                    pendingIntent
                )

            } else {

                alarmManager.setAndAllowWhileIdle(
                    AlarmManager.RTC_WAKEUP,
                    triggerTime,
                    pendingIntent
                )
            }

        } else {

            alarmManager.setExactAndAllowWhileIdle(
                AlarmManager.RTC_WAKEUP,
                triggerTime,
                pendingIntent
            )
        }
    }

    private fun calculateNotificationTime(
        eventDateTime: LocalDateTime,
        timing: String
    ): LocalDateTime {

        return when (timing) {

            "10 minutes before" ->
                eventDateTime.minusMinutes(10)

            "30 minutes before" ->
                eventDateTime.minusMinutes(30)

            "1 hour before" ->
                eventDateTime.minusHours(1)

            "2 hours before" ->
                eventDateTime.minusHours(2)

            "1 day before" ->
                eventDateTime.minusDays(1)

            "3 days before" ->
                eventDateTime.minusDays(3)

            "1 week before" ->
                eventDateTime.minusWeeks(1)

            else ->
                eventDateTime
        }
    }

    /*
     * Returns exactly one next occurrence.
     *
     * This is used by "Done for today" and by
     * the automatic recurrence receiver.
     */
    fun getNextOccurrence(
        reminder: Reminder
    ): Reminder {

        if (!isRecurring(reminder.repeatRule)) {
            return reminder
        }

        val currentDate =
            parseDate(reminder.date)
                ?: return reminder

        val anchorDate =
            resolveAnchorDate(reminder)
                ?: currentDate

        val nextDate =
            when (reminder.repeatRule) {

                "Daily" -> {
                    currentDate.plusDays(1)
                }

                "Weekly" -> {
                    currentDate.plusWeeks(1)
                }

                "Monthly" -> {

                    val nextMonth =
                        YearMonth
                            .from(currentDate)
                            .plusMonths(1)

                    val desiredDay =
                        anchorDate.dayOfMonth

                    val actualDay =
                        minOf(
                            desiredDay,
                            nextMonth.lengthOfMonth()
                        )

                    LocalDate.of(
                        nextMonth.year,
                        nextMonth.monthValue,
                        actualDay
                    )
                }

                "Yearly" -> {

                    val nextYear =
                        currentDate.year + 1

                    val anchorMonth =
                        anchorDate.monthValue

                    val targetMonth =
                        YearMonth.of(
                            nextYear,
                            anchorMonth
                        )

                    val desiredDay =
                        anchorDate.dayOfMonth

                    val actualDay =
                        minOf(
                            desiredDay,
                            targetMonth.lengthOfMonth()
                        )

                    LocalDate.of(
                        nextYear,
                        anchorMonth,
                        actualDay
                    )
                }

                else -> {
                    currentDate
                }
            }

        val repeatUntilDate =
            parseRepeatUntilDate(reminder)

        /*
         * Returning the same reminder is deliberate.
         * Existing callers already use the date change to
         * determine whether a recurrence was advanced.
         */
        if (
            repeatUntilDate != null &&
            nextDate.isAfter(repeatUntilDate)
        ) {
            return reminder
        }

        return reminder.copy(
            date =
                nextDate.format(
                    dateFormatter
                ),

            repeatAnchorDate =
                anchorDate.format(
                    dateFormatter
                )
        )
    }

    /**
     * True when another occurrence exists without crossing
     * the optional Repeat Until date.
     */
    fun hasNextOccurrence(
        reminder: Reminder
    ): Boolean {

        if (!isRecurring(reminder.repeatRule)) {
            return false
        }

        val nextReminder =
            getNextOccurrence(reminder)

        return nextReminder.date !=
                reminder.date
    }

    /**
     * True when the current occurrence is the last allowed
     * occurrence in a limited recurring series.
     */
    fun isFinalRecurringOccurrence(
        reminder: Reminder
    ): Boolean {

        return isRecurring(reminder.repeatRule) &&
                !hasNextOccurrence(reminder)
    }

    /*
     * Used after a phone restart.
     *
     * If several occurrences were missed while
     * the phone was off, keep advancing until
     * the next event is genuinely in the future.
     */
    fun moveRecurringReminderToFuture(
        reminder: Reminder
    ): Reminder {

        if (!isRecurring(reminder.repeatRule)) {
            return reminder
        }

        var workingReminder =
            ensureAnchorDate(reminder)

        var eventDateTime =
            parseEventDateTime(
                workingReminder
            ) ?: return workingReminder

        val now =
            LocalDateTime.now()

        var safetyCounter = 0

        while (
            !eventDateTime.isAfter(now) &&
            safetyCounter < 10000
        ) {

            val nextReminder =
                getNextOccurrence(
                    workingReminder
                )

            if (
                nextReminder.date ==
                workingReminder.date
            ) {
                break
            }

            workingReminder =
                nextReminder

            eventDateTime =
                parseEventDateTime(
                    workingReminder
                ) ?: break

            safetyCounter++
        }

        return workingReminder
    }

    private fun ensureAnchorDate(
        reminder: Reminder
    ): Reminder {

        if (!isRecurring(reminder.repeatRule)) {
            return reminder
        }

        if (
            reminder.repeatAnchorDate != "None" &&
            parseDate(
                reminder.repeatAnchorDate
            ) != null
        ) {
            return reminder
        }

        return reminder.copy(
            repeatAnchorDate =
                reminder.date
        )
    }

    private fun resolveAnchorDate(
        reminder: Reminder
    ): LocalDate? {

        if (
            reminder.repeatAnchorDate != "None"
        ) {

            val storedAnchor =
                parseDate(
                    reminder.repeatAnchorDate
                )

            if (storedAnchor != null) {
                return storedAnchor
            }
        }

        return parseDate(
            reminder.date
        )
    }

    private fun parseRepeatUntilDate(
        reminder: Reminder
    ): LocalDate? {

        if (
            reminder.repeatUntilDate.isBlank() ||
            reminder.repeatUntilDate == "None"
        ) {
            return null
        }

        return parseDate(
            reminder.repeatUntilDate
        )
    }

    private fun isOccurrenceWithinRepeatLimit(
        reminder: Reminder
    ): Boolean {

        val repeatUntilDate =
            parseRepeatUntilDate(reminder)
                ?: return true

        val occurrenceDate =
            parseDate(reminder.date)
                ?: return false

        return !occurrenceDate
            .isAfter(repeatUntilDate)
    }

    private fun parseDate(
        value: String
    ): LocalDate? {

        return try {

            LocalDate.parse(
                value,
                dateFormatter
            )

        } catch (
            exception: Exception
        ) {

            null
        }
    }

    private fun parseDeadlineDateTime(
        reminder: Reminder
    ): LocalDateTime? {

        if (
            reminder.deadlineDate.isBlank() ||
            reminder.deadlineDate == "None" ||
            reminder.deadlineTime.isBlank() ||
            reminder.deadlineTime == "None"
        ) {
            return null
        }

        return try {

            val date =
                LocalDate.parse(
                    reminder.deadlineDate,
                    dateFormatter
                )

            val time =
                LocalTime.parse(
                    reminder.deadlineTime,
                    timeFormatter
                )

            LocalDateTime.of(
                date,
                time
            )

        } catch (
            exception: Exception
        ) {

            null
        }
    }

    private fun parseEventDateTime(
        reminder: Reminder
    ): LocalDateTime? {

        return try {

            val date =
                LocalDate.parse(
                    reminder.date,
                    dateFormatter
                )

            val time =
                LocalTime.parse(
                    reminder.time,
                    timeFormatter
                )

            LocalDateTime.of(
                date,
                time
            )

        } catch (
            exception: Exception
        ) {

            null
        }
    }

    fun isRecurring(
        repeatRule: String
    ): Boolean {

        return repeatRule ==
                "Daily" ||
                repeatRule ==
                "Weekly" ||
                repeatRule ==
                "Monthly" ||
                repeatRule ==
                "Yearly"
    }

    fun cancelReminder(
        context: Context,
        reminderId: Int
    ) {

        val alarmManager =
            context.getSystemService(
                Context.ALARM_SERVICE
            ) as AlarmManager

        val notificationManager =
            context.getSystemService(
                Context.NOTIFICATION_SERVICE
            ) as NotificationManager

        timingCodes.forEach { (timing, _) ->

            val alarmId =
                createAlarmId(
                    reminderId,
                    timing
                )

            val reminderIntent =
                Intent(
                    context,
                    ReminderReceiver::class.java
                )

            val normalPendingIntent =
                PendingIntent.getBroadcast(
                    context,
                    alarmId,
                    reminderIntent,
                    PendingIntent.FLAG_NO_CREATE or
                            PendingIntent.FLAG_IMMUTABLE
                )

            if (
                normalPendingIntent != null
            ) {

                alarmManager.cancel(
                    normalPendingIntent
                )

                normalPendingIntent.cancel()
            }

            val snoozeAlarmId =
                createSnoozeAlarmRequestCode(
                    alarmId
                )

            val snoozeIntent =
                Intent(
                    context,
                    ReminderReceiver::class.java
                )

            val snoozePendingIntent =
                PendingIntent.getBroadcast(
                    context,
                    snoozeAlarmId,
                    snoozeIntent,
                    PendingIntent.FLAG_NO_CREATE or
                            PendingIntent.FLAG_IMMUTABLE
                )

            if (
                snoozePendingIntent != null
            ) {

                alarmManager.cancel(
                    snoozePendingIntent
                )

                snoozePendingIntent.cancel()
            }

            notificationManager.cancel(
                alarmId
            )
        }

        escalationTimingCodes.forEach { (timing, _) ->

            val alarmId =
                createEscalationAlarmId(
                    reminderId,
                    timing
                )

            val reminderIntent =
                Intent(
                    context,
                    ReminderReceiver::class.java
                )

            val pendingIntent =
                PendingIntent.getBroadcast(
                    context,
                    alarmId,
                    reminderIntent,
                    PendingIntent.FLAG_NO_CREATE or
                        PendingIntent.FLAG_IMMUTABLE
                )

            if (pendingIntent != null) {

                alarmManager.cancel(
                    pendingIntent
                )

                pendingIntent.cancel()
            }

            val snoozeAlarmId =
                createSnoozeAlarmRequestCode(
                    alarmId
                )

            val snoozeIntent =
                Intent(
                    context,
                    ReminderReceiver::class.java
                )

            val snoozePendingIntent =
                PendingIntent.getBroadcast(
                    context,
                    snoozeAlarmId,
                    snoozeIntent,
                    PendingIntent.FLAG_NO_CREATE or
                        PendingIntent.FLAG_IMMUTABLE
                )

            if (snoozePendingIntent != null) {

                alarmManager.cancel(
                    snoozePendingIntent
                )

                snoozePendingIntent.cancel()
            }

            notificationManager.cancel(
                alarmId
            )
        }

        val recurrenceIntent =
            Intent(
                context,
                RecurrenceReceiver::class.java
            )

        val recurrencePendingIntent =
            PendingIntent.getBroadcast(
                context,
                createRecurrenceAlarmId(
                    reminderId
                ),
                recurrenceIntent,
                PendingIntent.FLAG_NO_CREATE or
                        PendingIntent.FLAG_IMMUTABLE
            )

        if (
            recurrencePendingIntent != null
        ) {

            alarmManager.cancel(
                recurrencePendingIntent
            )

            recurrencePendingIntent.cancel()
        }

        val legacyIntent =
            Intent(
                context,
                ReminderReceiver::class.java
            )

        val legacyPendingIntent =
            PendingIntent.getBroadcast(
                context,
                reminderId,
                legacyIntent,
                PendingIntent.FLAG_NO_CREATE or
                        PendingIntent.FLAG_IMMUTABLE
            )

        if (
            legacyPendingIntent != null
        ) {

            alarmManager.cancel(
                legacyPendingIntent
            )

            legacyPendingIntent.cancel()
        }

        notificationManager.cancel(
            reminderId
        )
    }

    private fun createAlarmId(
        reminderId: Int,
        timing: String
    ): Int {

        val timingCode =
            timingCodes[timing] ?: 1

        return reminderId * 100 +
                timingCode
    }

    private fun createEscalationAlarmId(
        reminderId: Int,
        timing: String
    ): Int {

        val timingCode =
            escalationTimingCodes[timing] ?: 21

        return reminderId * 100 +
            timingCode
    }

    private fun createRecurrenceAlarmId(
        reminderId: Int
    ): Int {

        return reminderId * 100 +
                RECURRENCE_CODE
    }

    fun createSnoozeAlarmRequestCode(
        notificationId: Int
    ): Int {

        return notificationId * 10 + 9
    }
}

class ReminderReceiver : BroadcastReceiver() {

    override fun onReceive(
        context: Context,
        intent: Intent
    ) {

        val reminderId =
            intent.getIntExtra(
                "REMINDER_ID",
                0
            )

        val notificationId =
            intent.getIntExtra(
                "NOTIFICATION_ID",
                reminderId
            )

        val title =
            intent.getStringExtra(
                "REMINDER_TITLE"
            ) ?: "Reminder"

        val message =
            intent.getStringExtra(
                "REMINDER_MESSAGE"
            ) ?: "You have something to remember."

        val repeatRule =
            intent.getStringExtra(
                "REMINDER_REPEAT_RULE"
            ) ?: "None"

        val occurrenceDate =
            intent.getStringExtra(
                "OCCURRENCE_DATE"
            )

        val recurring =
            ReminderScheduler.isRecurring(
                repeatRule
            )

        createNotificationChannel(
            context
        )

        if (
            Build.VERSION.SDK_INT >=
            Build.VERSION_CODES.TIRAMISU &&
            ContextCompat.checkSelfPermission(
                context,
                Manifest.permission.POST_NOTIFICATIONS
            ) != PackageManager.PERMISSION_GRANTED
        ) {
            return
        }

        val openAppIntent =
            Intent(
                context,
                MainActivity::class.java
            ).apply {

                flags =
                    Intent.FLAG_ACTIVITY_NEW_TASK or
                            Intent.FLAG_ACTIVITY_CLEAR_TOP
            }

        val openAppPendingIntent =
            PendingIntent.getActivity(
                context,
                notificationId,
                openAppIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or
                        PendingIntent.FLAG_IMMUTABLE
            )

        val doneIntent =
            if (recurring) {

                Intent(
                    context,
                    DoneForTodayReceiver::class.java
                ).apply {

                    putExtra(
                        "REMINDER_ID",
                        reminderId
                    )

                    putExtra(
                        "OCCURRENCE_DATE",
                        occurrenceDate
                    )
                }

            } else {

                Intent(
                    context,
                    DoneReceiver::class.java
                ).apply {

                    putExtra(
                        "REMINDER_ID",
                        reminderId
                    )
                }
            }

        val donePendingIntent =
            PendingIntent.getBroadcast(
                context,
                notificationId * 10 + 1,
                doneIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or
                        PendingIntent.FLAG_IMMUTABLE
            )

        val snoozeIntent =
            Intent(
                context,
                SnoozeReceiver::class.java
            ).apply {

                putExtra(
                    "REMINDER_ID",
                    reminderId
                )

                putExtra(
                    "NOTIFICATION_ID",
                    notificationId
                )

                putExtra(
                    "REMINDER_TITLE",
                    title
                )

                putExtra(
                    "REMINDER_MESSAGE",
                    message
                )

                putExtra(
                    "REMINDER_REPEAT_RULE",
                    repeatRule
                )

                putExtra(
                    "OCCURRENCE_DATE",
                    occurrenceDate
                )
            }

        val snoozePendingIntent =
            PendingIntent.getBroadcast(
                context,
                notificationId * 10 + 2,
                snoozeIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or
                        PendingIntent.FLAG_IMMUTABLE
            )

        val doneButtonText =
            if (recurring) {
                "Done for today"
            } else {
                "Done"
            }

        val notification =
            NotificationCompat.Builder(
                context,
                CHANNEL_ID
            )
                .setSmallIcon(
                    android.R.drawable.ic_dialog_info
                )
                .setContentTitle(
                    title
                )
                .setContentText(
                    message
                )
                .setStyle(
                    NotificationCompat
                        .BigTextStyle()
                        .bigText(message)
                )
                .setPriority(
                    NotificationCompat.PRIORITY_HIGH
                )
                .setContentIntent(
                    openAppPendingIntent
                )
                .addAction(
                    0,
                    doneButtonText,
                    donePendingIntent
                )
                .addAction(
                    0,
                    "Snooze 10 min",
                    snoozePendingIntent
                )
                .setAutoCancel(true)
                .build()

        val notificationManager =
            context.getSystemService(
                Context.NOTIFICATION_SERVICE
            ) as NotificationManager

        notificationManager.notify(
            notificationId,
            notification
        )
    }

    private fun createNotificationChannel(
        context: Context
    ) {

        if (
            Build.VERSION.SDK_INT >=
            Build.VERSION_CODES.O
        ) {

            val channel =
                NotificationChannel(
                    CHANNEL_ID,
                    "Reminder Notifications",
                    NotificationManager.IMPORTANCE_HIGH
                ).apply {

                    description =
                        "Notifications for scheduled reminders"
                }

            val notificationManager =
                context.getSystemService(
                    Context.NOTIFICATION_SERVICE
                ) as NotificationManager

            notificationManager
                .createNotificationChannel(
                    channel
                )
        }
    }

    companion object {

        const val CHANNEL_ID =
            "personal_reminder_channel"
    }
}

import android.app.AlarmManager
import android.app.NotificationManager
import android.app.PendingIntent
import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.os.Build

class SnoozeReceiver : BroadcastReceiver() {

    override fun onReceive(
        context: Context,
        intent: Intent
    ) {

        val reminderId =
            intent.getIntExtra(
                "REMINDER_ID",
                0
            )

        val notificationId =
            intent.getIntExtra(
                "NOTIFICATION_ID",
                reminderId
            )

        val title =
            intent.getStringExtra(
                "REMINDER_TITLE"
            ) ?: "Reminder"

        val message =
            intent.getStringExtra(
                "REMINDER_MESSAGE"
            ) ?: "You have something to remember."

        val repeatRule =
            intent.getStringExtra(
                "REMINDER_REPEAT_RULE"
            ) ?: "None"

        val occurrenceDate =
            intent.getStringExtra(
                "OCCURRENCE_DATE"
            )

        val notificationManager =
            context.getSystemService(
                Context.NOTIFICATION_SERVICE
            ) as NotificationManager

        notificationManager.cancel(
            notificationId
        )

        val snoozeAlarmRequestCode =
            ReminderScheduler
                .createSnoozeAlarmRequestCode(
                    notificationId
                )

        val reminderIntent =
            Intent(
                context,
                ReminderReceiver::class.java
            ).apply {

                putExtra(
                    "REMINDER_ID",
                    reminderId
                )

                putExtra(
                    "NOTIFICATION_ID",
                    notificationId
                )

                putExtra(
                    "REMINDER_TITLE",
                    title
                )

                putExtra(
                    "REMINDER_MESSAGE",
                    "Snoozed reminder • $message"
                )

                putExtra(
                    "REMINDER_REPEAT_RULE",
                    repeatRule
                )

                putExtra(
                    "OCCURRENCE_DATE",
                    occurrenceDate
                )
            }

        val pendingIntent =
            PendingIntent.getBroadcast(
                context,
                snoozeAlarmRequestCode,
                reminderIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or
                        PendingIntent.FLAG_IMMUTABLE
            )

        val triggerTime =
            System.currentTimeMillis() +
                    (10 * 60 * 1000L)

        val alarmManager =
            context.getSystemService(
                Context.ALARM_SERVICE
            ) as AlarmManager

        if (
            Build.VERSION.SDK_INT >=
            Build.VERSION_CODES.S
        ) {

            if (
                alarmManager.canScheduleExactAlarms()
            ) {

                alarmManager.setExactAndAllowWhileIdle(
                    AlarmManager.RTC_WAKEUP,
                    triggerTime,
                    pendingIntent
                )

            } else {

                alarmManager.setAndAllowWhileIdle(
                    AlarmManager.RTC_WAKEUP,
                    triggerTime,
                    pendingIntent
                )
            }

        } else {

            alarmManager.setExactAndAllowWhileIdle(
                AlarmManager.RTC_WAKEUP,
                triggerTime,
                pendingIntent
            )
        }
    }
}

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.launch

class DoneReceiver : BroadcastReceiver() {

    override fun onReceive(
        context: Context,
        intent: Intent
    ) {

        val reminderId =
            intent.getIntExtra(
                "REMINDER_ID",
                0
            )

        if (reminderId == 0) {
            return
        }

        val pendingResult =
            goAsync()

        val scope =
            CoroutineScope(
                SupervisorJob() +
                        Dispatchers.IO
            )

        scope.launch {

            try {

                val database =
                    ReminderDatabase
                        .getDatabase(
                            context
                        )

                /*
                 * Mark the whole reminder series
                 * as completed.
                 */
                database
                    .reminderDao()
                    .markReminderCompleted(
                        reminderId
                    )

                /*
                 * Cancel every normal,
                 * recurring and snoozed alarm.
                 */
                ReminderScheduler
                    .cancelReminder(
                        context,
                        reminderId
                    )

            } finally {

                pendingResult.finish()
            }
        }
    }
}

@Dao

    suspend fun insertReminder(reminder: Reminder): Long

    suspend fun deleteReminder(reminder: Reminder)

    fun getAllReminders(): Flow<List<Reminder>>

    @Query("SELECT * FROM reminders WHERE isCompleted = 0 ORDER BY date ASC, time ASC")
    fun getActiveReminders(): Flow<List<Reminder>>

    @Query("SELECT * FROM reminders WHERE isCompleted = 1 ORDER BY createdAt DESC")
    fun getCompletedReminders(): Flow<List<Reminder>>

    suspend fun getReminderById(reminderId: Int): Reminder?

    suspend fun markReminderCompleted(reminderId: Int)

        """
        UPDATE reminders
        SET date = :newDate,
            repeatAnchorDate = :anchorDate
        WHERE id = :reminderId
        """
    )
    suspend fun updateReminderOccurrence(
        reminderId: Int,
        newDate: String,
        anchorDate: String
    )
