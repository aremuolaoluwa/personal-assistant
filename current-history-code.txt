data class MainDestination(
    val route: String,
    val label: String,
    val icon: ImageVector
)

@Composable
fun PersonalReminderApp() {

    val context =
        LocalContext.current

    val database =
        remember {

            ReminderDatabase
                .getDatabase(
                    context
                )
        }

    val coroutineScope =
        rememberCoroutineScope()

    var currentScreen
            by remember {

                mutableStateOf(
                    "home"
                )
            }

    var selectedReminder
            by remember {

                mutableStateOf<Reminder?>(
                    null
                )
            }

    val activeReminders
            by database
                .reminderDao()
                .getActiveReminders()
                .collectAsState(
                    initial =
                        emptyList()
                )

    val completedReminders
            by database
                .reminderDao()
                .getCompletedReminders()
                .collectAsState(
                    initial =
                        emptyList()
                )

    val destinations =
        listOf(

            MainDestination(
                route = "home",
                label = "Home",
                icon =
                    Icons.Outlined.Home
            ),

            MainDestination(
                route = "schedule",
                label = "Schedule",
                icon =
                    Icons.Outlined.EventNote
            ),

            MainDestination(
                route = "history",
                label = "History",
                icon =
                    Icons.Outlined.History
            ),

            MainDestination(
                route = "settings",
                label = "Settings",
                icon =
                    Icons.Outlined.Settings
            )
        )

    val mainRoutes =
        destinations
            .map {
                it.route
            }
            .toSet()

    fun completeReminder(
        reminder: Reminder
    ) {

        if (
            ReminderScheduler
                .isRecurring(
                    reminder.repeatRule
                )
        ) {

            coroutineScope.launch {

                ReminderScheduler
                    .cancelReminder(
                        context,
                        reminder.id
                    )

                val nextReminder =
                    ReminderScheduler
                        .getNextOccurrence(
                            reminder
                        )

                if (
                    nextReminder.date ==
                    reminder.date
                ) {

                    /*
                     * This was the final allowed occurrence.
                     * Completing it finishes the recurring series.
                     */
                    database
                        .reminderDao()
                        .updateReminder(
                            reminder.copy(
                                isCompleted = true,
                                repeatRule = "None",
                                repeatAnchorDate = "None",
                                repeatUntilDate = "None"
                            )
                        )

                } else {

                    database
                        .reminderDao()
                        .updateReminderOccurrence(
                            reminderId =
                                reminder.id,

                            newDate =
                                nextReminder.date,

                            anchorDate =
                                nextReminder
                                    .repeatAnchorDate
                        )

                    ReminderScheduler
                        .scheduleReminder(
                            context,
                            nextReminder
                        )
                }
            }

        } else {

            ReminderScheduler
                .cancelReminder(
                    context,
                    reminder.id
                )

            coroutineScope.launch {

                database
                    .reminderDao()
                    .updateReminder(
                        reminder.copy(
                            isCompleted =
                                true
                        )
                    )
            }
        }
    }

    fun deleteReminder(
        reminder: Reminder
    ) {

        ReminderScheduler
            .cancelReminder(
                context,
                reminder.id
            )

        coroutineScope.launch {

            database
                .reminderDao()
                .deleteReminder(
                    reminder
                )
        }
    }

    fun endRepeat(
        reminder: Reminder
    ) {

        coroutineScope.launch {

            ReminderScheduler
                .cancelReminder(
                    context,
                    reminder.id
                )

            val oneOffReminder =
                reminder.copy(
                    repeatRule =
                        "None",

                    repeatAnchorDate =
                        "None",

                    repeatUntilDate =
                        "None"
                )

            database
                .reminderDao()
                .updateReminder(
                    oneOffReminder
                )

            ReminderScheduler
                .scheduleReminder(
                    context,
                    oneOffReminder
                )
        }
    }

    fun restoreReminder(
        reminder: Reminder
    ) {

        coroutineScope.launch {

            var restoredReminder =
                reminder.copy(
                    isCompleted =
                        false
                )

            restoredReminder =
                ReminderScheduler
                    .moveRecurringReminderToFuture(
                        restoredReminder
                    )

            database
                .reminderDao()
                .updateReminder(
                    restoredReminder
                )

            ReminderScheduler
                .scheduleReminder(
                    context,
                    restoredReminder
                )
        }
    }

    Scaffold(

        modifier =
            Modifier.fillMaxSize(),

        containerColor =
            MaterialTheme
                .colorScheme
                .background,

        bottomBar = {

            if (
                currentScreen in
                mainRoutes
            ) {

                NavigationBar(
                    containerColor =
                        MaterialTheme
                            .colorScheme
                            .surface
                ) {

                    destinations
                        .forEach {
                                destination ->

                            NavigationBarItem(

                                selected =
                                    currentScreen ==
                                            destination.route,

                                onClick = {

                                    currentScreen =
                                        destination.route

                                    selectedReminder =
                                        null
                                },

                                icon = {

                                    Icon(
                                        imageVector =
                                            destination.icon,

                                        contentDescription =
                                            destination.label
                                    )
                                },

                                label = {

                                    Text(
                                        text =
                                            destination.label
                                    )
                                }
                            )
                        }
                }
            }
        },

        floatingActionButton = {

            if (
                currentScreen ==
                "home" ||
                currentScreen ==
                "schedule"
            ) {

                ExtendedFloatingActionButton(

                    onClick = {

                        currentScreen =
                            "addReminder"
                    },

                    icon = {

                        Icon(
                            imageVector =
                                Icons.Outlined.Add,

                            contentDescription =
                                "Add reminder"
                        )
                    },

                    text = {

                        Text(
                            text =
                                "Add reminder"
                        )
                    }
                )
            }
        }

    ) { innerPadding ->

        when (
            currentScreen
        ) {

            "home" -> {

                DashboardScreen(

                    reminders =
                        activeReminders,

                    completedCount =
                        completedReminders
                            .size,

                    onEdit = {
                            reminder ->

                        selectedReminder =
                            reminder

                        currentScreen =
                            "editReminder"
                    },

                    onDone = {
                            reminder ->

                        completeReminder(
                            reminder
                        )
                    },

                    onDelete = {
                            reminder ->

                        deleteReminder(
                            reminder
                        )
                    },

                    onEndRepeat = {
                            reminder ->

                        endRepeat(
                            reminder
                        )
                    },

                    onOpenSchedule = {

                        currentScreen =
                            "schedule"
                    },

                    modifier =
                        Modifier.padding(
                            innerPadding
                        )
                )
            }

            "schedule" -> {

                ScheduleScreen(

                    reminders =
                        activeReminders,

                    onEdit = {
                            reminder ->

                        selectedReminder =
                            reminder

                        currentScreen =
                            "editReminder"
                    },

                    onDone = {
                            reminder ->

                        completeReminder(
                            reminder
                        )
                    },

                    onDelete = {
                            reminder ->

                        deleteReminder(
                            reminder
                        )
                    },

                    onEndRepeat = {
                            reminder ->

                        endRepeat(
                            reminder
                        )
                    },

                    modifier =
                        Modifier.padding(
                            innerPadding
                        )
                )
            }

            "history" -> {

                HistoryScreen(

                    reminders =
                        completedReminders,

                    onRestore = {
                            reminder ->

                        restoreReminder(
                            reminder
                        )
                    },

                    onDelete = {
                            reminder ->

                        deleteReminder(
                            reminder
                        )
                    },

                    modifier =
                        Modifier.padding(
                            innerPadding
                        )
                )
            }

            "settings" -> {

                SettingsScreen(
                    modifier =
                        Modifier.padding(
                            innerPadding
                        )
                )
            }

            "addReminder" -> {

                AddReminderScreen(

                    onBack = {

                        currentScreen =
                            "home"
                    },

                    modifier =
                        Modifier.padding(
                            innerPadding
                        )
                )
            }

            "editReminder" -> {

                selectedReminder
                    ?.let {
                            reminder ->

                        EditReminderScreen(

                            reminder =
                                reminder,

                            onBack = {

                                selectedReminder =
                                    null

                                currentScreen =
                                    "home"
                            },

                            modifier =
                                Modifier.padding(
                                    innerPadding
                                )
                        )
                    }
            }
        }
    }
}

@Composable
fun DashboardScreen(
    reminders: List<Reminder>,
    completedCount: Int,
    onEdit: (Reminder) -> Unit,
    onDone: (Reminder) -> Unit,
    onDelete: (Reminder) -> Unit,
    onEndRepeat: (Reminder) -> Unit,
    onOpenSchedule: () -> Unit,
    modifier: Modifier = Modifier
) {

    val today =
        LocalDate.now()

    val sortedReminders =
        reminders.sortedWith(
            compareBy(
                {
                    parseReminderDate(
                        it.date
                    ) ?: LocalDate.MAX
                },
                {
                    it.time
                }
            )
        )

    val todayReminders =
        sortedReminders
            .filter {

                parseReminderDate(
                    it.date
                ) == today
            }

    val overdueReminders =
        sortedReminders
            .filter { reminder ->

                val dueDate =
                    effectiveDueDate(
                        reminder
                    )

                dueDate != null &&
                        dueDate.isBefore(
                            today
                        )
            }

    val allAttentionReminders =
        sortedReminders
            .filter { reminder ->

                val overdue =
                    effectiveDueDate(
                        reminder
                    )?.isBefore(
                        today
                    ) == true

                overdue ||
                        reminder.priority ==
                        "Critical" ||
                        reminder.priority ==
                        "Important"
            }

    val attentionReminders =
        allAttentionReminders
            .take(4)

    val approachingDeadlines =
        reminders
            .filter { reminder ->

                val dueDate =
                    effectiveDueDate(
                        reminder
                    )

                reminder.type.equals(
                    "Deadline",
                    ignoreCase = true
                ) &&
                        dueDate != null &&
                        !dueDate.isBefore(
                            today
                        )
            }
            .sortedBy { reminder ->
                effectiveDueDate(
                    reminder
                ) ?: LocalDate.MAX
            }
            .take(5)

    val recurringReminders =
        sortedReminders
            .filter {

                ReminderScheduler
                    .isRecurring(
                        it.repeatRule
                    )
            }
            .take(4)

    val criticalCount =
        reminders.count {

            it.priority ==
                    "Critical"
        }

    val greeting =
        when (
            LocalTime.now().hour
        ) {

            in 5..11 ->
                "Good morning"

            in 12..16 ->
                "Good afternoon"

            else ->
                "Good evening"
        }

    LazyColumn(

        modifier =
            modifier
                .fillMaxSize(),

        contentPadding =
            androidx.compose.foundation.layout.PaddingValues(
                start = 20.dp,
                end = 20.dp,
                top = 20.dp,
                bottom = 110.dp
            ),

        verticalArrangement =
            Arrangement.spacedBy(
                18.dp
            )

    ) {

        item {

            Column {

                Text(
                    text =
                        greeting,

                    style =
                        MaterialTheme
                            .typography
                            .bodyLarge,

                    color =
                        MaterialTheme
                            .colorScheme
                            .onSurfaceVariant
                )

                Spacer(
                    modifier =
                        Modifier.height(
                            4.dp
                        )
                )

                Text(
                    text =
                        "Your day at a glance",

                    style =
                        MaterialTheme
                            .typography
                            .headlineMedium
                )
            }
        }

        item {

            DashboardSummaryCard(

                attentionCount =
                    allAttentionReminders
                        .size,

                todayCount =
                    todayReminders
                        .size,

                criticalCount =
                    criticalCount,

                overdueCount =
                    overdueReminders
                        .size
            )
        }

        item {

            DashboardSectionHeader(

                title =
                    "Needs your attention",

                subtitle =
                    "Important, critical and overdue items"
            )
        }

        if (
            attentionReminders
                .isEmpty()
        ) {

            item {

                DashboardEmptyCard(
                    message =
                        "Nothing urgent right now."
                )
            }

        } else {

            items(
                items =
                    attentionReminders,

                key = {
                    "attention_${it.id}"
                }

            ) {
                    reminder ->

                DashboardReminderCard(

                    reminder =
                        reminder,

                    onEdit = {
                        onEdit(
                            reminder
                        )
                    },

                    onDone = {
                        onDone(
                            reminder
                        )
                    },

                    onDelete = {
                        onDelete(
                            reminder
                        )
                    },

                    onEndRepeat = {
                        onEndRepeat(
                            reminder
                        )
                    }
                )
            }
        }

        item {

            DashboardSectionHeader(

                title =
                    "Today",

                subtitle =
                    if (
                        todayReminders
                            .isEmpty()
                    ) {

                        "Your day is clear"

                    } else {

                        "${todayReminders.size} item" +
                                if (
                                    todayReminders
                                        .size == 1
                                ) {
                                    ""
                                } else {
                                    "s"
                                } +
                                " scheduled"
                    }
            )
        }

        if (
            todayReminders
                .isEmpty()
        ) {

            item {

                DashboardEmptyCard(
                    message =
                        "No reminders scheduled for today."
                )
            }

        } else {

            items(
                items =
                    todayReminders,

                key = {
                    "today_${it.id}"
                }

            ) {
                    reminder ->

                DashboardReminderCard(

                    reminder =
                        reminder,

                    onEdit = {
                        onEdit(
                            reminder
                        )
                    },

                    onDone = {
                        onDone(
                            reminder
                        )
                    },

                    onDelete = {
                        onDelete(
                            reminder
                        )
                    },

                    onEndRepeat = {
                        onEndRepeat(
                            reminder
                        )
                    }
                )
            }
        }

        item {

            DashboardSectionHeader(

                title =
                    "Approaching deadlines",

                subtitle =
                    "Keep important dates visible",

                actionText =
                    "View schedule",

                onAction =
                    onOpenSchedule
            )
        }

        if (
            approachingDeadlines
                .isEmpty()
        ) {

            item {

                DashboardEmptyCard(
                    message =
                        "No upcoming deadlines."
                )
            }

        } else {

            items(
                items =
                    approachingDeadlines,

                key = {
                    "deadline_${it.id}"
                }

            ) {
                    reminder ->

                DeadlineRow(
                    reminder =
                        reminder,

                    onClick = {
                        onEdit(
                            reminder
                        )
                    }
                )
            }
        }

        if (
            recurringReminders
                .isNotEmpty()
        ) {

            item {

                DashboardSectionHeader(

                    title =
                        "Ongoing commitments",

                    subtitle =
                        "Reminders that keep returning"
                )
            }

            items(
                items =
                    recurringReminders,

                key = {
                    "recurring_${it.id}"
                }

            ) {
                    reminder ->

                OngoingReminderRow(
                    reminder =
                        reminder,

                    onClick = {
                        onEdit(
                            reminder
                        )
                    }
                )
            }
        }

        item {

            DashboardProgressCard(
                completedCount =
                    completedCount
            )
        }
    }
}

@Composable
fun DashboardSectionHeader(
    title: String,
    subtitle: String,
    actionText: String? = null,
    onAction: (() -> Unit)? = null
) {

    Row(
        modifier =
            Modifier.fillMaxWidth(),

        horizontalArrangement =
            Arrangement.SpaceBetween,

        verticalAlignment =
            Alignment.Bottom
    ) {

        Column(
            modifier =
                Modifier.weight(
                    1f
                )
        ) {

            Text(
                text =
                    title,

                style =
                    MaterialTheme
                        .typography
                        .titleLarge
            )

            Spacer(
                modifier =
                    Modifier.height(
                        2.dp
                    )
            )

            Text(
                text =
                    subtitle,

                style =
                    MaterialTheme
                        .typography
                        .bodyMedium,

                color =
                    MaterialTheme
                        .colorScheme
                        .onSurfaceVariant
            )
        }

        if (
            actionText != null &&
            onAction != null
        ) {

            TextButton(
                onClick =
                    onAction
            ) {

                Text(
                    text =
                        actionText
                )
            }
        }
    }
}

@Composable
fun DashboardReminderCard(
    reminder: Reminder,
    onEdit: () -> Unit,
    onDone: () -> Unit,
    onDelete: () -> Unit,
    onEndRepeat: () -> Unit
) {

    val recurring =
        ReminderScheduler
            .isRecurring(
                reminder.repeatRule
            )

    val priorityColor =
        priorityColor(
            reminder.priority
        )

    val isDeadline =
        reminder.type.equals(
            "Deadline",
            ignoreCase = true
        ) &&
                reminder.deadlineDate != "None" &&
                reminder.deadlineDate.isNotBlank()

    val finalDeadlineDate =
        if (isDeadline) {
            parseReminderDate(
                reminder.deadlineDate
            )
        } else {
            null
        }

    val deadlineDays =
        finalDeadlineDate?.let { date ->
            ChronoUnit.DAYS.between(
                LocalDate.now(),
                date
            )
        }

    val nextAlertDateTime =
        nextScheduledAlertDateTime(
            reminder
        )

    val deadlineOccurrenceDiffers =
        isDeadline &&
                (
                    reminder.date !=
                            reminder.deadlineDate ||
                            reminder.time !=
                            reminder.deadlineTime
                    )

    val showOccurrence =
        !isDeadline ||
                recurring ||
                deadlineOccurrenceDiffers

    val occurrenceLabel =
        when {
            isDeadline && recurring ->
                "Next recurring reminder"

            isDeadline ->
                "Reminder schedule"

            reminder.type.equals(
                "Event",
                ignoreCase = true
            ) ->
                if (recurring) {
                    "Next event"
                } else {
                    "Event"
                }

            reminder.type.equals(
                "Task",
                ignoreCase = true
            ) ->
                if (recurring) {
                    "Next task"
                } else {
                    "Task"
                }

            recurring ->
                "Next occurrence"

            else ->
                "Scheduled for"
        }

    ElevatedCard(

        modifier =
            Modifier.fillMaxWidth(),

        shape =
            RoundedCornerShape(
                22.dp
            ),

        colors =
            CardDefaults
                .elevatedCardColors(
                    containerColor =
                        MaterialTheme
                            .colorScheme
                            .surface
                )
    ) {

        Column(
            modifier =
                Modifier.padding(
                    18.dp
                )
        ) {

            Row(
                modifier =
                    Modifier.fillMaxWidth(),

                horizontalArrangement =
                    Arrangement.SpaceBetween,

                verticalAlignment =
                    Alignment.Top
            ) {

                Row(
                    modifier =
                        Modifier.weight(
                            1f
                        ),

                    verticalAlignment =
                        Alignment.CenterVertically
                ) {

                    Box(
                        modifier =
                            Modifier
                                .size(
                                    10.dp
                                )
                                .background(
                                    color =
                                        priorityColor,

                                    shape =
                                        CircleShape
                                )
                    )

                    Spacer(
                        modifier =
                            Modifier.size(
                                10.dp
                            )
                    )

                    Text(
                        text =
                            reminder.title,

                        style =
                            MaterialTheme
                                .typography
                                .titleMedium,

                        maxLines =
                            2,

                        overflow =
                            TextOverflow.Ellipsis,

                        modifier =
                            Modifier.weight(
                                1f
                            )
                    )
                }

                Spacer(
                    modifier =
                        Modifier.size(
                            8.dp
                        )
                )

                PriorityChip(
                    priority =
                        reminder.priority
                )
            }

            if (showOccurrence) {

                Spacer(
                    modifier =
                        Modifier.height(
                            14.dp
                        )
                )

                Text(
                    text =
                        occurrenceLabel,

                    style =
                        MaterialTheme
                            .typography
                            .labelLarge,

                    color =
                        MaterialTheme
                            .colorScheme
                            .onSurfaceVariant
                )

                Spacer(
                    modifier =
                        Modifier.height(
                            2.dp
                        )
                )

                Text(
                    text =
                        "${relativeReminderDateLabel(reminder.date)} • ${reminder.time}",

                    style =
                        MaterialTheme
                            .typography
                            .bodyLarge,

                    fontWeight =
                        FontWeight.SemiBold
                )
            }

            if (isDeadline) {

                Spacer(
                    modifier =
                        Modifier.height(
                            14.dp
                        )
                )

                Surface(
                    modifier =
                        Modifier.fillMaxWidth(),

                    shape =
                        RoundedCornerShape(
                            16.dp
                        ),

                    color =
                        MaterialTheme
                            .colorScheme
                            .primaryContainer
                            .copy(
                                alpha = 0.55f
                            )
                ) {

                    Row(
                        modifier =
                            Modifier
                                .fillMaxWidth()
                                .padding(
                                    14.dp
                                ),

                        horizontalArrangement =
                            Arrangement.SpaceBetween,

                        verticalAlignment =
                            Alignment.CenterVertically
                    ) {

                        Column(
                            modifier =
                                Modifier.weight(
                                    1f
                                )
                        ) {

                            Text(
                                text =
                                    "Final deadline",

                                style =
                                    MaterialTheme
                                        .typography
                                        .labelLarge,

                                color =
                                    MaterialTheme
                                        .colorScheme
                                        .primary
                            )

                            Spacer(
                                modifier =
                                    Modifier.height(
                                        3.dp
                                    )
                            )

                            Text(
                                text =
                                    formatReminderDate(
                                        reminder.deadlineDate
                                    ) +
                                            if (
                                                reminder.deadlineTime != "None" &&
                                                reminder.deadlineTime.isNotBlank()
                                            ) {
                                                " • ${reminder.deadlineTime}"
                                            } else {
                                                ""
                                            },

                                style =
                                    MaterialTheme
                                        .typography
                                        .bodyLarge,

                                fontWeight =
                                    FontWeight.SemiBold
                            )
                        }

                        if (deadlineDays != null) {

                            Spacer(
                                modifier =
                                    Modifier.size(
                                        10.dp
                                    )
                            )

                            Surface(
                                shape =
                                    RoundedCornerShape(
                                        50.dp
                                    ),

                                color =
                                    if (
                                        deadlineDays <= 3
                                    ) {
                                        CriticalRed.copy(
                                            alpha = 0.12f
                                        )
                                    } else {
                                        MaterialTheme
                                            .colorScheme
                                            .surface
                                    }
                            ) {

                                Text(
                                    text =
                                        deadlineText(
                                            deadlineDays
                                        ),

                                    style =
                                        MaterialTheme
                                            .typography
                                            .labelLarge,

                                    color =
                                        if (
                                            deadlineDays <= 3
                                        ) {
                                            CriticalRed
                                        } else {
                                            MaterialTheme
                                                .colorScheme
                                                .primary
                                        },

                                    modifier =
                                        Modifier.padding(
                                            horizontal = 10.dp,
                                            vertical = 6.dp
                                        )
                                )
                            }
                        }
                    }
                }
            }

            Spacer(
                modifier =
                    Modifier.height(
                        14.dp
                    )
            )

            Text(
                text =
                    "Next alert",

                style =
                    MaterialTheme
                        .typography
                        .labelLarge,

                color =
                    MaterialTheme
                        .colorScheme
                        .onSurfaceVariant
            )

            Spacer(
                modifier =
                    Modifier.height(
                        2.dp
                    )
            )

            Text(
                text =
                    if (
                        nextAlertDateTime != null
                    ) {
                        formatAlertDateTime(
                            nextAlertDateTime
                        )
                    } else {
                        "No future alert scheduled"
                    },

                style =
                    MaterialTheme
                        .typography
                        .bodyLarge,

                fontWeight =
                    FontWeight.SemiBold,

                color =
                    if (
                        nextAlertDateTime != null
                    ) {
                        MaterialTheme
                            .colorScheme
                            .onSurface
                    } else {
                        MaterialTheme
                            .colorScheme
                            .onSurfaceVariant
                    }
            )

            Spacer(
                modifier =
                    Modifier.height(
                        12.dp
                    )
            )

            Row(
                horizontalArrangement =
                    Arrangement.spacedBy(
                        8.dp
                    )
            ) {

                SmallInfoChip(
                    text =
                        reminder.type
                )

                if (
                    reminder.category
                        .isNotBlank()
                ) {

                    SmallInfoChip(
                        text =
                            reminder.category
                    )
                }

                if (recurring) {

                    SmallInfoChip(
                        text =
                            reminder.repeatRule,

                        icon =
                            Icons
                                .Outlined
                                .Repeat
                    )
                }
            }

            if (recurring) {

                Spacer(
                    modifier =
                        Modifier.height(
                            10.dp
                        )
                )

                Text(
                    text =
                        recurrenceSummaryText(
                            reminder
                        ),

                    style =
                        MaterialTheme
                            .typography
                            .bodyMedium,

                    color =
                        MaterialTheme
                            .colorScheme
                            .onSurfaceVariant

            }

                reminder.notes
                    .isNotBlank()
            ) {

                    modifier =
                        Modifier.height(
                            12.dp
                        )
                )

                Text(
                    text =
                        reminder.notes,

                    style =
                        MaterialTheme
                            .typography
                            .bodyMedium,

                    color =
                        MaterialTheme
                            .colorScheme
                            .onSurfaceVariant,

                    maxLines =
                        2,

                    overflow =
                        TextOverflow.Ellipsis
                )
            }

            Spacer(
                modifier =
                    Modifier.height(
                        16.dp
                    )
            )

            Row(
                modifier =
                    Modifier.fillMaxWidth(),

                horizontalArrangement =
                    Arrangement.spacedBy(
                        10.dp
                    )
            ) {

                FilledTonalButton(
                    onClick =
                        onDone,

                    modifier =
                        Modifier.weight(
                            1f
                        )
                ) {

                    Icon(
                        imageVector =
                            Icons
                                .Outlined
                                .CheckCircle,

                        contentDescription =
                            null,

                        modifier =
                            Modifier.size(
                                18.dp
                            )
                    )

                    Spacer(
                        modifier =
                            Modifier.size(
                                6.dp
                            )
                    )

                    Text(
                        text =
                            if (recurring) {
                                "Done today"
                            } else {
                                "Done"
                            }
                    )
                }

                OutlinedButton(
                    onClick =
                        onEdit,

                    modifier =
                        Modifier.weight(
                            1f
                        )
                ) {

                    Icon(
                        imageVector =
                            Icons
                                .Outlined
                                .Edit,

                        contentDescription =
                            null,

                        modifier =
                            Modifier.size(
                                18.dp
                            )

                        modifier =

                            Modifier.size(
                                4.dp
                            )
                    )

                    Text(
                        text =
                            "Delete"
                    )
                }
            }
        }
    }
}

@Composable
fun DeadlineRow(
    reminder: Reminder,
    onClick: () -> Unit
) {

    val date =
        effectiveDueDate(
            reminder
        )

    val deadlineTime =
        if (
            reminder.deadlineTime != "None" &&
            reminder.deadlineTime.isNotBlank()
        ) {
            reminder.deadlineTime
        } else {
            reminder.time
        }

    val today =
        LocalDate.now()

    val days =
        if (date != null) {

            ChronoUnit.DAYS
                .between(
                    today,
                    date
                )

        } else {

            0L
        }

    ElevatedCard(

        onClick =
            onClick,

        modifier =
            Modifier.fillMaxWidth(),

        shape =
            RoundedCornerShape(
                18.dp
            )
    ) {

        Row(
            modifier =
                Modifier
                    .fillMaxWidth()
                    .padding(
                        16.dp
                    ),

            verticalAlignment =
                Alignment.CenterVertically
        ) {

            Surface(
                shape =
                    RoundedCornerShape(
                        14.dp
                    ),

                color =
                    MaterialTheme
                        .colorScheme
                        .primaryContainer
            ) {

                Column(
                    modifier =
                        Modifier.padding(
                            horizontal =
                                12.dp,

                            vertical =
                                8.dp
                        ),

                    horizontalAlignment =
                        Alignment.CenterHorizontally
                ) {

                    Text(
                        text =
                            date
                                ?.dayOfMonth
                                ?.toString()
                                ?: "--",

                        style =
                            MaterialTheme
                                .typography
                                .titleLarge
                    )

                    Text(
                        text =
                            date
                                ?.month
                                ?.name
                                ?.take(
                                    3
                                )
                                ?: "",

                        style =
                            MaterialTheme
                                .typography
                                .bodyMedium
                    )
                }
            }

            Spacer(
                modifier =
                    Modifier.size(
                        14.dp
                    )
            )

            Column(
                modifier =
                    Modifier.weight(
                        1f
                    )
            ) {

                Text(
                    text =
                        reminder.title,

                    style =
                        MaterialTheme
                            .typography
                            .titleMedium,

                    maxLines =
                        1,

                    overflow =
                        TextOverflow.Ellipsis
                )

                Spacer(
                    modifier =
                        Modifier.height(
                            3.dp
                        )
                )

                Text(
                    text =
                        "${deadlineText(days)} • $deadlineTime",

                    style =
                        MaterialTheme
                            .typography
                            .bodyMedium,

                    color =
                        if (
                            days <= 3
                        ) {

                            CriticalRed

                        } else {

                            MaterialTheme
                                .colorScheme
                                .onSurfaceVariant
                        }
                )
            }

            PriorityChip(
                priority =
                    reminder.priority
            )
        }
    }
}

@Composable
fun ScheduleScreen(
    reminders: List<Reminder>,
    onEdit: (Reminder) -> Unit,
    onDone: (Reminder) -> Unit,
    onDelete: (Reminder) -> Unit,
    onEndRepeat: (Reminder) -> Unit,
    modifier: Modifier = Modifier
) {

    val sorted =
        reminders.sortedWith(
            compareBy(
                {
                    parseReminderDate(
                        it.date
                    ) ?: LocalDate.MAX
                },
                {
                    it.time
                }
            )
        )

    LazyColumn(

        modifier =
            modifier
                .fillMaxSize(),

        contentPadding =
            androidx.compose.foundation.layout.PaddingValues(
                start = 20.dp,
                end = 20.dp,
                top = 22.dp,
                bottom = 110.dp
            ),

        verticalArrangement =
            Arrangement.spacedBy(
                14.dp
            )

    ) {

        item {

            Column {

                Text(
                    text =
                        "Schedule",

                    style =
                        MaterialTheme
                            .typography
                            .headlineMedium
                )

                Text(
                    text =
                        "Everything coming up, in one place.",

                    style =
                        MaterialTheme
                            .typography
                            .bodyLarge,

                    color =
                        MaterialTheme
                            .colorScheme
                            .onSurfaceVariant
                )
            }
        }

        if (
            sorted.isEmpty()
        ) {

            item {

                DashboardEmptyCard(
                    message =
                        "Nothing scheduled yet."
                )
            }

        } else {

            items(
                items =
                    sorted,

                key = {
                    it.id
                }

            ) {
                    reminder ->

                DashboardReminderCard(

                    reminder =
                        reminder,

                    onEdit = {
                        onEdit(
                            reminder
                        )
                    },

                    onDone = {
                        onDone(
                            reminder
                        )
                    },

                    onDelete = {
                        onDelete(
                            reminder
                        )
                    },

                    onEndRepeat = {
                        onEndRepeat(
                            reminder
                        )
                    }
                )
            }
        }
    }
}

private fun parseReminderDate(
    value: String
): LocalDate? {

    return try {

        LocalDate.parse(
            value,
            DateTimeFormatter
                .ofPattern(
                    "dd/MM/yyyy"
                )
        )

    } catch (
        exception: Exception
    ) {

        null
    }
}

private fun formatReminderDate(
    value: String
): String {

    val date =
        parseReminderDate(
            value
        )

    return if (
        date != null
    ) {

        date.format(
            DateTimeFormatter
                .ofPattern(
                    "d MMM yyyy",
                    Locale.UK
                )
        )

    } else {

        value
    }
}

private fun relativeReminderDateLabel(
    value: String
): String {

    val date =
        parseReminderDate(
            value
        ) ?: return value

    val days =
        ChronoUnit.DAYS.between(
            LocalDate.now(),
            date
        )

    return when (days) {

        0L ->
            "Today"

        1L ->
            "Tomorrow"

        else ->
            formatReminderDate(
                value
            )
    }
}

private fun recurrenceSummaryText(
    reminder: Reminder
): String {

    val repeatName =
        reminder.repeatRule
            .lowercase(
                Locale.UK
            )

    val repeatUntil =
        reminder.repeatUntilDate

    return if (
        repeatUntil != "None" &&
        repeatUntil.isNotBlank() &&
        parseReminderDate(
            repeatUntil
        ) != null
    ) {

        "Repeats $repeatName until ${formatReminderDate(repeatUntil)}"

    } else {

        "Repeats $repeatName"
    }
}

private fun effectiveDueDate(
    reminder: Reminder
): LocalDate? {

    if (
        reminder.type.equals(
            "Deadline",
            ignoreCase = true
        ) &&
        reminder.deadlineDate != "None" &&
        reminder.deadlineDate.isNotBlank()
    ) {

        val finalDeadline =
            parseReminderDate(
                reminder.deadlineDate
            )

        if (finalDeadline != null) {
            return finalDeadline
        }
    }

    return parseReminderDate(
        reminder.date
    )
}

private fun nextScheduledAlertDateTime(
    reminder: Reminder
): LocalDateTime? {

    val eventDate =
        parseReminderDate(
            reminder.date
        ) ?: return null

    val eventTime =
        try {

            LocalTime.parse(
                reminder.time,
                DateTimeFormatter
                    .ofPattern(
                        "HH:mm"
                    )
            )

        } catch (
            exception: Exception
        ) {

            return null
        }

    val eventDateTime =
        LocalDateTime.of(
            eventDate,
            eventTime
        )

    val now =
        LocalDateTime.now()

    return parseReminderTimings(
        reminder.reminderTiming
    )
        .map { timing ->

            when (timing) {

                "10 minutes before" ->
                    eventDateTime
                        .minusMinutes(
                            10
                        )

                "30 minutes before" ->
                    eventDateTime
                        .minusMinutes(
                            30
                        )

                "1 hour before" ->
                    eventDateTime
                        .minusHours(
                            1
                        )

                "2 hours before" ->
                    eventDateTime
                        .minusHours(
                            2
                        )

                "1 day before" ->
                    eventDateTime
                        .minusDays(
                            1
                        )

                "3 days before" ->
                    eventDateTime
                        .minusDays(
                            3
                        )

                "1 week before" ->
                    eventDateTime
                        .minusWeeks(
                            1
                        )

                else ->
                    eventDateTime
            }
        }
        .filter {
            it.isAfter(
                now
            )
        }
        .minOrNull()
}

private fun formatAlertDateTime(
    value: LocalDateTime
): String {

    val storedDate =
        value
            .toLocalDate()
            .format(
                DateTimeFormatter
                    .ofPattern(
                        "dd/MM/yyyy"
                    )
            )

    val time =
        value
            .toLocalTime()
            .format(
                DateTimeFormatter
                    .ofPattern(
                        "HH:mm"
                    )
            )

    return "${relativeReminderDateLabel(storedDate)} • $time"
}

private fun deadlineText(
    days: Long
): String {

    return when {

        days < 0 -> {

            val overdueDays =
                -days

            if (
                overdueDays == 1L
            ) {
                "Overdue by 1 day"
            } else {
                "Overdue by $overdueDays days"
            }
        }

        days == 0L ->
            "Due today"

        days == 1L ->
            "Due tomorrow"

        else ->
            "$days days left"
    }
}

    priority: String
): Color {

    return when (
        priority
    ) {

        "Critical" ->
            CriticalRed

        "Important" ->
            ImportantAmber

        else ->
            Color(
                0xFF0B43E6
            )
    }

@Database(

        Reminder::class,
        ReminderAction::class
    ],
    version = 5,
    exportSchema = false
)
abstract class ReminderDatabase : RoomDatabase() {

    abstract fun reminderDao(): ReminderDao

    abstract fun reminderActionDao(): ReminderActionDao

    companion object {

        @Volatile
        private var INSTANCE: ReminderDatabase? = null

        /*
         * Version 1 -> 2
         * Added repeatRule.
         */
        private val MIGRATION_1_2 =
            object : Migration(1, 2) {

                override fun migrate(
                    database: SupportSQLiteDatabase
                ) {

                    database.execSQL(
                        """
                        ALTER TABLE reminders
                        ADD COLUMN repeatRule TEXT NOT NULL DEFAULT 'None'
                        """.trimIndent()
                    )
                }
            }

        /*
         * Version 2 -> 3
         * Added recurrence anchor.
         */
        private val MIGRATION_2_3 =
            object : Migration(2, 3) {

                override fun migrate(
                    database: SupportSQLiteDatabase
                ) {

                    database.execSQL(
                        """
                        ALTER TABLE reminders
                        ADD COLUMN repeatAnchorDate TEXT NOT NULL DEFAULT 'None'
                        """.trimIndent()
                    )

                    database.execSQL(
                        """
                        UPDATE reminders
                        SET repeatAnchorDate = date
                        WHERE repeatRule != 'None'
                        """.trimIndent()
                    )
                }
            }

        /*
         * Version 3 -> 4
         * Added intelligent reminder fields
         * and action history.
         */
        private val MIGRATION_3_4 =
            object : Migration(3, 4) {

                override fun migrate(
                    database: SupportSQLiteDatabase
                ) {

                    database.execSQL(
                        """
                        ALTER TABLE reminders
                        ADD COLUMN deadlineDate TEXT NOT NULL DEFAULT 'None'
                        """.trimIndent()
                    )

                    database.execSQL(
                        """
                        ALTER TABLE reminders
                        ADD COLUMN deadlineTime TEXT NOT NULL DEFAULT 'None'
                        """.trimIndent()
                    )

                    database.execSQL(
                        """
                        ALTER TABLE reminders

                        """.trimIndent()
                    )

                    database.execSQL(
                        """
                        ALTER TABLE reminders
                        ADD COLUMN followUpUntilCompleted INTEGER NOT NULL DEFAULT 0
                        """.trimIndent()
                    )

                    database.execSQL(
                        """
                        ALTER TABLE reminders

                        """.trimIndent()
                    )

                    database.execSQL(
                        """
                        ALTER TABLE reminders
                        ADD COLUMN snoozeCount INTEGER NOT NULL DEFAULT 0
                        """.trimIndent()
                    )

                    database.execSQL(
                        """
                        ALTER TABLE reminders
                        ADD COLUMN lastSnoozedAt INTEGER DEFAULT NULL
                        """.trimIndent()
                    )

                    database.execSQL(
                        """
                        CREATE TABLE IF NOT EXISTS reminder_actions (
                            id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL,
                            reminderId INTEGER NOT NULL,
                            actionType TEXT NOT NULL,
                            timestamp INTEGER NOT NULL,
                            occurrenceDate TEXT,
                            details TEXT,
                            FOREIGN KEY(reminderId)
                                REFERENCES reminders(id)
                                ON UPDATE NO ACTION
                                ON DELETE CASCADE
                        )
                        """.trimIndent()
                    )

                    database.execSQL(
                        """
                        CREATE INDEX IF NOT EXISTS
                        index_reminder_actions_reminderId
                        ON reminder_actions(reminderId)
                        """.trimIndent()
                    )
                }
            }

        /*
         * Version 4 -> 5
         *
         * Repair migration.
         *
         * Some installations reached database version 4
         * before the Reminder entity contained all of the
         * new intelligent-reminder fields.
         *
         * This migration checks the actual database and
         * adds only columns that are missing. Existing
         * reminders are preserved.
         */
        private val MIGRATION_4_5 =
            object : Migration(4, 5) {

                override fun migrate(
                    database: SupportSQLiteDatabase
                ) {

                    val columns =
                        mutableSetOf<String>()

                    database
                        .query(
                            "PRAGMA table_info(reminders)"
                        )
                        .use { cursor ->

                            val nameIndex =
                                cursor.getColumnIndex(
                                    "name"
                                )

                            while (
                                cursor.moveToNext()
                            ) {

                                columns.add(
                                    cursor.getString(
                                        nameIndex
                                    )
                                )
                            }
                        }

                    fun addColumnIfMissing(
                        columnName: String,
                        sql: String
                    ) {

                        if (
                            columnName !in columns
                        ) {

                            database.execSQL(
                                sql
                            )

                            columns.add(
                                columnName
                            )
                        }
                    }

                    val anchorWasMissing =
                        "repeatAnchorDate" !in columns

                    addColumnIfMissing(
                        "repeatRule",
                        """
                        ALTER TABLE reminders
                        ADD COLUMN repeatRule TEXT NOT NULL DEFAULT 'None'
                        """.trimIndent()
                    )

                    addColumnIfMissing(
                        "repeatAnchorDate",
                        """
                        ALTER TABLE reminders
                        ADD COLUMN repeatAnchorDate TEXT NOT NULL DEFAULT 'None'
                        """.trimIndent()
                    )

                    addColumnIfMissing(
                        "deadlineDate",
                        """
                        ALTER TABLE reminders
                        ADD COLUMN deadlineDate TEXT NOT NULL DEFAULT 'None'
                        """.trimIndent()
                    )

                    addColumnIfMissing(
                        "deadlineTime",
                        """
                        ALTER TABLE reminders
                        ADD COLUMN deadlineTime TEXT NOT NULL DEFAULT 'None'
                        """.trimIndent()
                    )

                    addColumnIfMissing(
                        "repeatUntilDate",
                        """
                        ALTER TABLE reminders
                        ADD COLUMN repeatUntilDate TEXT NOT NULL DEFAULT 'None'
                        """.trimIndent()
                    )

                    addColumnIfMissing(
                        "followUpUntilCompleted",
                        """
                        ALTER TABLE reminders
                        ADD COLUMN followUpUntilCompleted INTEGER NOT NULL DEFAULT 0
                        """.trimIndent()
                    )

                    addColumnIfMissing(
                        "escalationEnabled",
                        """
                        ALTER TABLE reminders
                        ADD COLUMN escalationEnabled INTEGER NOT NULL DEFAULT 0
                        """.trimIndent()
                    )

                    addColumnIfMissing(
                        "snoozeCount",
                        """
                        ALTER TABLE reminders
                        ADD COLUMN snoozeCount INTEGER NOT NULL DEFAULT 0
                        """.trimIndent()
                    )

                    addColumnIfMissing(
                        "lastSnoozedAt",
                        """
                        ALTER TABLE reminders
                        ADD COLUMN lastSnoozedAt INTEGER DEFAULT NULL
                        """.trimIndent()
                    )

                    if (
                        anchorWasMissing
                    ) {

                        database.execSQL(
                            """
                            UPDATE reminders
                            SET repeatAnchorDate = date
                            WHERE repeatRule != 'None'
                            """.trimIndent()
                        )
                    }

                    database.execSQL(
                        """
                        CREATE TABLE IF NOT EXISTS reminder_actions (
                            id INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL,
                            reminderId INTEGER NOT NULL,
                            actionType TEXT NOT NULL,
                            timestamp INTEGER NOT NULL,
                            occurrenceDate TEXT,
                            details TEXT,
                            FOREIGN KEY(reminderId)
                                REFERENCES reminders(id)
                                ON UPDATE NO ACTION
                                ON DELETE CASCADE

                    )

                        index_reminder_actions_reminderId
                        ON reminder_actions(reminderId)
                        """.trimIndent()
                    )
                }
            }

        fun getDatabase(
            context: Context

                                MIGRATION_2_3,
                                MIGRATION_3_4,
                                MIGRATION_4_5
                            )
                            .build()

                    INSTANCE =
                        instance

                    instance
                }
        }
    }

class ReminderReceiver : BroadcastReceiver() {

    override fun onReceive(
        context: Context,
        intent: Intent
    ) {

        val reminderId =
            intent.getIntExtra(
                "REMINDER_ID",
                0
            )

        val notificationId =
            intent.getIntExtra(
                "NOTIFICATION_ID",
                reminderId
            )

        val title =
            intent.getStringExtra(
                "REMINDER_TITLE"
            ) ?: "Reminder"

        val message =
            intent.getStringExtra(
                "REMINDER_MESSAGE"
            ) ?: "You have something to remember."

        val repeatRule =
            intent.getStringExtra(
                "REMINDER_REPEAT_RULE"
            ) ?: "None"

        val occurrenceDate =
            intent.getStringExtra(
                "OCCURRENCE_DATE"
            )

        val followUpEnabled =
            intent.getBooleanExtra(
                "FOLLOW_UP_ENABLED",
                false
            )

        val priority =
            intent.getStringExtra(
                "REMINDER_PRIORITY"
            ) ?: "Normal"

        val reminderType =
            intent.getStringExtra(
                "REMINDER_TYPE"
            ) ?: "Reminder"

        val reminderTiming =
            intent.getStringExtra(
                "REMINDER_TIMING"
            ) ?: "At time of event"

        val isFollowUp =
            intent.getBooleanExtra(
                "IS_FOLLOW_UP",
                false
            )

        val recurring =
            ReminderScheduler.isRecurring(
                repeatRule
            )

        ensureNotificationChannels(
            context
        )

        if (
            Build.VERSION.SDK_INT >=
            Build.VERSION_CODES.TIRAMISU &&
            ContextCompat.checkSelfPermission(
                context,
                Manifest.permission.POST_NOTIFICATIONS
            ) != PackageManager.PERMISSION_GRANTED
        ) {
            return
        }

        val openAppIntent =
            Intent(
                context,
                MainActivity::class.java
            ).apply {

                flags =
                    Intent.FLAG_ACTIVITY_NEW_TASK or
                            Intent.FLAG_ACTIVITY_CLEAR_TOP
            }

        val openAppPendingIntent =
            PendingIntent.getActivity(
                context,
                notificationId,
                openAppIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or
                        PendingIntent.FLAG_IMMUTABLE
            )

        val doneIntent =
            if (recurring) {

                Intent(
                    context,
                    DoneForTodayReceiver::class.java
                ).apply {

                    putExtra(
                        "REMINDER_ID",
                        reminderId
                    )

                    putExtra(
                        "OCCURRENCE_DATE",
                        occurrenceDate
                    )
                }

            } else {

                Intent(
                    context,
                    DoneReceiver::class.java
                ).apply {

                    putExtra(
                        "REMINDER_ID",
                        reminderId
                    )
                }
            }

        val donePendingIntent =
            PendingIntent.getBroadcast(
                context,
                notificationId * 10 + 1,
                doneIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or
                        PendingIntent.FLAG_IMMUTABLE
            )

        val snoozeIntent =
            Intent(
                context,
                SnoozeReceiver::class.java
            ).apply {

                putExtra(
                    "REMINDER_ID",
                    reminderId
                )

                putExtra(
                    "NOTIFICATION_ID",
                    notificationId
                )

                putExtra(
                    "REMINDER_TITLE",
                    title
                )

                putExtra(
                    "REMINDER_MESSAGE",
                    message
                )

                putExtra(
                    "REMINDER_REPEAT_RULE",
                    repeatRule
                )

                putExtra(
                    "OCCURRENCE_DATE",
                    occurrenceDate
                )

                putExtra(
                    "FOLLOW_UP_ENABLED",
                    followUpEnabled
                )

                putExtra(
                    "REMINDER_PRIORITY",
                    priority
                )

                putExtra(
                    "REMINDER_TYPE",
                    reminderType
                )

                putExtra(
                    "REMINDER_TIMING",
                    reminderTiming
                )

                putExtra(
                    "IS_FOLLOW_UP",
                    isFollowUp
                )
            }

        val snoozePendingIntent =
            PendingIntent.getBroadcast(
                context,
                notificationId * 10 + 2,
                snoozeIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or
                        PendingIntent.FLAG_IMMUTABLE
            )

        val stillWorkingIntent =
            Intent(
                context,
                StillWorkingReceiver::class.java
            ).apply {

                putExtra(
                    "REMINDER_ID",
                    reminderId
                )

                putExtra(
                    "NOTIFICATION_ID",
                    notificationId
                )
            }

        val stillWorkingPendingIntent =
            PendingIntent.getBroadcast(
                context,
                notificationId * 10 + 3,
                stillWorkingIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or
                    PendingIntent.FLAG_IMMUTABLE
            )

        val isEventAdvanceAlert =
            reminderType.equals(
                "Event",
                ignoreCase = true
            ) &&
                !isFollowUp &&
                reminderTiming !=
                    "At time of event"

        val doneButtonText =
            when {

                recurring ->
                    "Done for today"

                reminderType.equals(
                    "Task",
                    ignoreCase = true
                ) ||
                    reminderType.equals(
                        "Deadline",
                        ignoreCase = true
                    ) ->
                    "Completed"

                else ->
                    "Done"
            }

        val keepActiveButtonText =
            when {

                reminderType.equals(
                    "Event",
                    ignoreCase = true
                ) ->
                    "Still relevant"

                reminderType.equals(
                    "Reminder",
                    ignoreCase = true
                ) ->
                    "Keep active"

                else ->
                    "Still working"
            }

            NotificationCompat.Builder(
                context,
                channelIdForPriority(
                    priority
                )
            )
                .setSmallIcon(
                    android.R.drawable.ic_dialog_info
                )
                .setContentTitle(
                    title
                )
                .setContentText(
                    message
                )
                .setStyle(
                    NotificationCompat
                        .BigTextStyle()
                        .bigText(message)
                )
                .setPriority(
                    compatPriorityForPriority(
                        priority
                    )
                )
                .setCategory(
                    NotificationCompat.CATEGORY_REMINDER
                )
                .setContentIntent(
                    openAppPendingIntent
                )
                .setAutoCancel(true)

        /*
         * Event alerts that fire before the actual event do not
         * show a completion button. Tapping "Done" several days
         * early could otherwise complete the whole event or move
         * a recurring event to its next occurrence.
         */
        if (!isEventAdvanceAlert) {

            notificationBuilder.addAction(
                0,
                doneButtonText,
                donePendingIntent
            )
        }

        if (
            followUpEnabled &&
            !recurring &&
            !isEventAdvanceAlert

                0,
                keepActiveButtonText,

            0,
            "Snooze 10 min",
            snoozePendingIntent
        )

        val notification =
            notificationBuilder.build()

        val notificationManager =
            context.getSystemService(
                Context.NOTIFICATION_SERVICE
            ) as NotificationManager

        notificationManager.notify(
            notificationId,
            notification
        )
    }

        const val CHANNEL_ID =
            "personal_reminder_channel"

        private const val NORMAL_CHANNEL_ID =
            "personal_reminder_normal_v2"

        private const val IMPORTANT_CHANNEL_ID =
            "personal_reminder_important_v2"

        private const val CRITICAL_CHANNEL_ID =
            "personal_reminder_critical_v2"

        fun channelIdForPriority(
            priority: String
        ): String {

            return when (priority) {

                "Critical" ->
                    CRITICAL_CHANNEL_ID

                "Important" ->
                    IMPORTANT_CHANNEL_ID

                else ->
                    NORMAL_CHANNEL_ID
            }
        }

        fun compatPriorityForPriority(
            priority: String
        ): Int {

            return when (priority) {

                "Critical" ->
                    NotificationCompat.PRIORITY_MAX

                "Important" ->
                    NotificationCompat.PRIORITY_HIGH

                else ->
                    NotificationCompat.PRIORITY_DEFAULT
            }
        }

        fun ensureNotificationChannels(
            context: Context
        ) {

            if (
                Build.VERSION.SDK_INT >=
                Build.VERSION_CODES.O
            ) {

                val normalChannel =
                    NotificationChannel(
                        NORMAL_CHANNEL_ID,
                        "Normal reminders",
                        NotificationManager.IMPORTANCE_DEFAULT
                    ).apply {

                        description =
                            "Standard reminders and follow-ups"
                    }

                val importantChannel =
                    NotificationChannel(
                        IMPORTANT_CHANNEL_ID,
                        "Important reminders",
                        NotificationManager.IMPORTANCE_HIGH
                    ).apply {

                        description =
                            "Important reminders that need stronger attention"
                    }

                val criticalChannel =
                    NotificationChannel(
                        CRITICAL_CHANNEL_ID,
                        "Critical reminders",
                        NotificationManager.IMPORTANCE_HIGH
                    ).apply {

                        description =
                            "Critical reminders and urgent follow-ups"
                    }

                val notificationManager =
                    context.getSystemService(
                        Context.NOTIFICATION_SERVICE
                    ) as NotificationManager

                notificationManager
                    .createNotificationChannels(
                        listOf(
                            normalChannel,
                            importantChannel,
                            criticalChannel
                        )
                    )
            }
        }
    }

                0
            )

        val notificationId =
            intent.getIntExtra(
                "NOTIFICATION_ID",
                reminderId
            )

        val title =
            intent.getStringExtra(
                "REMINDER_TITLE"
            ) ?: "Reminder"

        val message =
            intent.getStringExtra(
                "REMINDER_MESSAGE"
            ) ?: "You have something to remember."

        val repeatRule =
            intent.getStringExtra(
                "REMINDER_REPEAT_RULE"
            ) ?: "None"

        val occurrenceDate =
            intent.getStringExtra(
                "OCCURRENCE_DATE"
            )

        val followUpEnabled =
            intent.getBooleanExtra(
                "FOLLOW_UP_ENABLED",
                false
            )

        val priority =
            intent.getStringExtra(
                "REMINDER_PRIORITY"
            ) ?: "Normal"

        val reminderType =
            intent.getStringExtra(
                "REMINDER_TYPE"
            ) ?: "Reminder"

        val reminderTiming =
            intent.getStringExtra(
                "REMINDER_TIMING"
            ) ?: "At time of event"

        val isFollowUp =
            intent.getBooleanExtra(
                "IS_FOLLOW_UP",
                false
            )

        val notificationManager =
            context.getSystemService(
                Context.NOTIFICATION_SERVICE
            ) as NotificationManager

        notificationManager.cancel(
            notificationId
        )

        val snoozeAlarmRequestCode =
            ReminderScheduler
                .createSnoozeAlarmRequestCode(
                    notificationId
                )

        val reminderIntent =
            Intent(
                context,
                ReminderReceiver::class.java
            ).apply {

                putExtra(
                    "REMINDER_ID",
                    reminderId
                )

                putExtra(
                    "NOTIFICATION_ID",
                    notificationId
                )

                putExtra(
                    "REMINDER_TITLE",
                    title
                )

                putExtra(
                    "REMINDER_MESSAGE",
                    "Snoozed reminder • $message"
                )

                putExtra(
                    "REMINDER_REPEAT_RULE",
                    repeatRule
                )

                putExtra(
                    "OCCURRENCE_DATE",
                    occurrenceDate
                )

                putExtra(
                    "FOLLOW_UP_ENABLED",
                    followUpEnabled
                )

                putExtra(
                    "REMINDER_PRIORITY",
                    priority
                )

                putExtra(
                    "REMINDER_TYPE",
                    reminderType
                )

                putExtra(
                    "REMINDER_TIMING",
                    reminderTiming
                )

                putExtra(
                    "IS_FOLLOW_UP",
                    isFollowUp
                )
            }

        val pendingIntent =
            PendingIntent.getBroadcast(
                context,
                snoozeAlarmRequestCode,
                reminderIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or
                        PendingIntent.FLAG_IMMUTABLE
            )

        val triggerTime =
            System.currentTimeMillis() +
                    (10 * 60 * 1000L)

        val alarmManager =
            context.getSystemService(
                Context.ALARM_SERVICE
            ) as AlarmManager

        if (
            Build.VERSION.SDK_INT >=
            Build.VERSION_CODES.S
        ) {

            if (
                alarmManager.canScheduleExactAlarms()
            ) {

                alarmManager.setExactAndAllowWhileIdle(
                    AlarmManager.RTC_WAKEUP,
                    triggerTime,
                    pendingIntent
                )

            } else {

                alarmManager.setAndAllowWhileIdle(
                    AlarmManager.RTC_WAKEUP,
                    triggerTime,
                    pendingIntent
                )
            }

        } else {

            alarmManager.setExactAndAllowWhileIdle(
                AlarmManager.RTC_WAKEUP,
                triggerTime,
                pendingIntent
            )
        }
    }
}

package com.josepharemu.reminderassistant

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob

class DoneForTodayReceiver : BroadcastReceiver() {

    override fun onReceive(
        context: Context,
        intent: Intent
    ) {

        val reminderId =
            intent.getIntExtra(
                "REMINDER_ID",
                0
            )

        val occurrenceDate =
            intent.getStringExtra(
                "OCCURRENCE_DATE"
            )

        if (reminderId == 0) {
            return
        }

        val pendingResult =
            goAsync()

        val scope =
            CoroutineScope(
                SupervisorJob() +
                        Dispatchers.IO
            )

        scope.launch {

            try {

                val database =
                    ReminderDatabase
                        .getDatabase(
                            context
                        )

                val dao =
                    database.reminderDao()

                val currentReminder =
                    dao.getReminderById(
                        reminderId
                    ) ?: return@launch

                if (
                    currentReminder.isCompleted ||
                    !ReminderScheduler.isRecurring(
                        currentReminder.repeatRule
                    )
                ) {

                    return@launch
                }

                /*
                 * Cancel all alarms first.
                 *
                 * If the automatic recurrence receiver
                 * already moved the reminder forward,
                 * we will simply schedule that new
                 * occurrence again below.
                 */
                ReminderScheduler
                    .cancelReminder(
                        context,
                        reminderId
                    )

                var reminderToSchedule =
                    currentReminder

                /*
                 * Only advance the date if this
                 * notification still belongs to the
                 * reminder's current occurrence.
                 *
                 * This prevents:
                 *
                 * 16 Oct -> 17 Oct -> 18 Oct
                 *
                 * when the automatic recurrence alarm
                 * had already changed it to 17 Oct.
                 */
                if (
                    occurrenceDate == null ||
                    currentReminder.date ==
                    occurrenceDate
                ) {

                    val nextReminder =
                        ReminderScheduler
                            .getNextOccurrence(
                                currentReminder
                            )

                    /*
                     * No date change means this is the final
                     * allowed occurrence. "Done for today"
                     * should therefore finish the series.
                     */
                    if (
                        nextReminder.date ==
                        currentReminder.date
                    ) {

                        dao.updateReminder(
                            currentReminder.copy(
                                isCompleted = true,
                                repeatRule = "None",
                                repeatAnchorDate = "None",
                                repeatUntilDate = "None"
                            )
                        )

                        return@launch
                    }

                    dao.updateReminderOccurrence(
                        reminderId =
                            reminderId,

                        newDate =
                            nextReminder.date,

                        anchorDate =
                            nextReminder.repeatAnchorDate
                    )

                    reminderToSchedule =
                        nextReminder
                }

                val latestReminder =
                    dao.getReminderById(
                        reminderId
                    )

                if (
                    latestReminder != null &&
                    !latestReminder.isCompleted &&
                    ReminderScheduler.isRecurring(
                        latestReminder.repeatRule
                    )
                ) {

                    reminderToSchedule =
                        latestReminder

                    ReminderScheduler
                        .scheduleReminder(
                            context,
                            reminderToSchedule
                        )
                }

            } finally {

                pendingResult.finish()
            }
        }
    }
}
